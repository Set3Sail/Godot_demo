# 轮盘世界 · 6 角色 Mechanic 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development
>
> **依赖:** Battle Framework Foundation (已完成)
> **Spec:** docs/superpowers/specs/2026-05-31-轮盘世界-战斗框架-prd.md §7

**Goal:** 实现剩余 6 个角色战斗机制（墨夜/温静/盛元/金英/红发/莉莲），全部继承 BaseMechanic。

**Architecture:** 每个角色一个独立脚本，继承 BaseMechanic，覆写对应生命周期回调。通过 BattleManager 在 `_setup_dependencies` 中加载当前角色 Mechanic 实例。

### Task 1: 墨夜 Mechanic（冰霜 + 天怒）

**Create:** `scripts/battle/mechanics/moye_mechanic.gd`

```gdscript
class_name MoyeMechanic

var _frost_energy_used: int = 0          # 本回合已触发次数

func on_battle_start() -> void:
    _fury_used = false
    _frost_energy_counter = 0

    _frost_energy_used = 0

        _fury_used = true

    if _status_mgr.has_status(source, "frost"):
        var frost_stacks: int = _status_mgr.get_stacks(source, "frost")
        # 伤害修正已在 get_damage_multiplier 中处理

func get_damage_multiplier(target: Node) -> float:
    var mult: float = 1.0
    # 天怒首次双倍
    if not _fury_used:
        # 检查当前卡是否有天怒标签 — 由外部 context 传入
        pass
    return mult

## 天怒标签卡牌伤害倍率（由 BattleManager 调用）
func get_fury_multiplier() -> float:
    if not _fury_used:
        return 1.0  # 首次使用前返回正常倍率
    return 1.0  # 已用后正常

## 首次天怒使用时返回 2.0
func try_activate_fury() -> float:
    if not _fury_used:
        _fury_used = true
        return 2.0
    return 1.0

## 冰晶护符: 每施加 3 层冰霜, +1 精神力
func on_frost_applied(stacks: int) -> void:
    if _frost_energy_used >= _frost_energy_max_per_turn:
        return
    _frost_energy_counter += stacks
    while _frost_energy_counter >= 3 and _frost_energy_used < _frost_energy_max_per_turn:
        _frost_energy_counter -= 3
        _frost_energy_used += 1
        if _resource_mgr:
            _resource_mgr.modify_player_energy(1)
```

### Task 2: 温静 Mechanic（标记 + 连击）

**Create:** `scripts/battle/mechanics/wenjing_mechanic.gd`

```gdscript
class_name WenjingMechanic
extends BaseMechanic

var _combo_counter: int = 0
var _combo_threshold: int = 3            # 连击阈值（遗物可降为 2）
var _mark_bonus_used_this_turn: bool = false  # 初始遗物: 每回合首次

func on_battle_start() -> void:
    _combo_counter = 0
    _mark_bonus_used_this_turn = false

func on_turn_start(_round_num: int) -> void:
    _mark_bonus_used_this_turn = false

func on_card_played(card: CardData, target: Node) -> void:
    if card.card_type == CardData.CardType.ATTACK:
        _combo_counter += 1
        if _combo_counter >= _combo_threshold:
            _trigger_combo()
            _combo_counter = 0

func get_extra_damage(target: Node) -> int:
    var extra: int = 0
    if _status_mgr.has_status(target, "mark"):
        extra += int(ceilf(extra * 0.25))  # +25% 伤害
    return extra

## 首次对被标记敌人造成伤害: 追加 3 伤（玫瑰印记）
func try_mark_bonus(target: Node) -> int:
    if _mark_bonus_used_this_turn:
        return 0
    if _status_mgr.has_status(target, "mark"):
        _mark_bonus_used_this_turn = true
        return 3
    return 0

## 引爆标记: 移除标记 + 造成额外伤害
func detonate_mark(target: Node, base_damage: int) -> int:
    if not _status_mgr.has_status(target, "mark"):
        return 0
    var remaining_turns: int = _status_mgr.get_stacks(target, "mark")
    _status_mgr.remove_status(target, "mark")
    return base_damage + remaining_turns * 5

func _trigger_combo() -> void:
    var pool: Array[String] = ["draw", "energy", "aoe"]
    var chosen: String = pool.pick_random()
    match chosen:
        "draw":
            if _card_mgr:
                _card_mgr.draw_cards(1)
        "energy":
            if _resource_mgr:
                _resource_mgr.modify_player_energy(1)
        "aoe":
            if _status_mgr:
                for enemy in _enemy_mgr.get_alive_enemies():
                    _status_mgr.thorn_damage.emit(enemy, 4)

## 遗物: 连击阈值降至 2
func set_combo_threshold(t: int) -> void:
    _combo_threshold = t
```

### Task 3: 盛元 Mechanic（复仇 + 血祭）

**Create:** `scripts/battle/mechanics/shengyuan_mechanic.gd`

```gdscript
class_name ShengyuanMechanic
extends BaseMechanic

var _revenge_stacks: int = 0

func on_battle_start() -> void:
    _revenge_stacks = 0

func on_damage_taken(_target: Node, _source: Node, amount: int) -> void:
    _revenge_stacks += floori(amount / 5.0)

func on_turn_start(_round_num: int) -> void:
    # 亡者意志: HP<50% → +1精神力; HP<25% → +2精神力
    if _resource_mgr:
        var hp: int = _resource_mgr.get_player_hp()
        var max_hp: int = _resource_mgr.get_player_max_hp()
        var ratio: float = float(hp) / float(max_hp)
        if ratio < 0.25:
            _resource_mgr.modify_player_energy(2)
        elif ratio < 0.5:
            _resource_mgr.modify_player_energy(1)

func get_extra_damage(_target: Node) -> int:
    var extra: int = _revenge_stacks * 2
    return extra

## 攻击牌打出后清零复仇
func on_card_played(card: CardData, _target: Node) -> void:
    if card.card_type == CardData.CardType.ATTACK:
        _revenge_stacks = 0

## 血祭: 卡牌 cost_hp 在 BattleManager.play_card 中处理
func get_blood_sacrifice_bonus(cost_hp: int) -> float:
    return 1.0 + (cost_hp * 0.1)  # 每点 HP 消耗 +10% 效果
```

### Task 4: 金英 Mechanic（神罚 + 亲和）

**Create:** `scripts/battle/mechanics/jinying_mechanic.gd`

```gdscript
class_name JinyingMechanic
extends BaseMechanic

var _holy_charge: int = 0              # 神罚蓄能
var _heal_tracker: int = 0             # 神圣之触计数器
var _damage_taken_this_turn: int = 0   # 亲和条件追踪

func on_battle_start() -> void:
    _holy_charge = 0
    _heal_tracker = 0
    _damage_taken_this_turn = 0

func on_turn_start(_round_num: int) -> void:
    _damage_taken_this_turn = 0

func on_heal(_target: Node, amount: int) -> void:
    _holy_charge += amount
    # 神圣之触: 每回复 4HP, 随机敌人 3 神圣伤害
    _heal_tracker += amount
    while _heal_tracker >= 4:
        _heal_tracker -= 4
        if _enemy_mgr:
            var enemies: Array = _enemy_mgr.get_alive_enemies()
            if not enemies.is_empty():
                var target: Node = enemies.pick_random()
                # 神圣伤害通过 RESOLVE 管道 — 由 EffectResolver 处理
                _status_mgr.thorn_damage.emit(target, 3)  # 复用 thorn_damage 信号

func on_damage_taken(_target: Node, _source: Node, amount: int) -> void:
    _damage_taken_this_turn += amount

func on_turn_end() -> void:
    # 亲和: 本回合未受伤 → 回复 2HP
    if _damage_taken_this_turn == 0 and _resource_mgr:
        _resource_mgr.modify_player_hp(2, "heal")
    # 遗物放宽: <=3 伤害也触发

## 消耗神罚蓄能转化为神圣伤害
func consume_holy_charge(amount: int) -> int:
    var consumed: int = mini(amount, _holy_charge)
    _holy_charge -= consumed
    return consumed

## 亲和条件放宽（遗物）
func set_affinity_threshold(max_damage: int) -> void:
    _affinity_threshold = max_damage
```

### Task 5: 红发 Mechanic（召唤 + 牺牲）

**Create:** `scripts/battle/mechanics/hongfa_mechanic.gd`

```gdscript
class_name HongfaMechanic
extends BaseMechanic

var _max_zombies: int = 4              # 最大丧尸数（遗物 +1 = 5）

func on_battle_start() -> void:
    pass

## 召唤丧尸
func summon_zombie(hp: int = 10, attack: int = 5) -> void:
    if _enemy_mgr.get_alive_allies().size() >= _max_zombies:
        return
    var zombie := Node.new()
    zombie.set_meta("display_name", "丧尸随从")
    zombie.set_meta("max_hp", hp)
    zombie.set_meta("attack_damage", attack)
    # 注册到 EnemyManager 友方格
    _enemy_mgr._allies.append(zombie)
    add_child(zombie)
    _enemy_mgr.ally_spawned.emit(zombie, _enemy_mgr._allies.find(zombie))

## 牺牲所有丧尸，每只对全体敌人造成伤害
func sacrifice_all_zombies() -> int:
    var allies: Array = _enemy_mgr.get_alive_allies()
    var count: int = allies.size()
    for ally in allies:
        _enemy_mgr._allies.erase(ally)
        _enemy_mgr.ally_killed.emit(ally)
        if is_instance_valid(ally):
            ally.queue_free()
    return count

## 丧尸死亡 → +2 格挡
func on_ally_killed() -> void:
    if _resource_mgr:
        _resource_mgr.modify_player_block(2)

## 意识结晶: 每只存活丧尸 +1 格挡/回合
func get_extra_block() -> int:
    var alive: int = _enemy_mgr.get_alive_allies().size()
    return mini(alive, 4)  # 最多 +4

## 遗物: 最大丧尸数 +1
func set_max_zombies(n: int) -> void:
    _max_zombies = n
```

### Task 6: 莉莲 Mechanic（战甲 + 过热）

**Create:** `scripts/battle/mechanics/lilian_mechanic.gd`

```gdscript
class_name LilianMechanic
extends BaseMechanic

enum ArmorMode { ATTACK, DEFENSE }

var _current_mode: ArmorMode = ArmorMode.ATTACK
var _switch_counter: int = 0
var _overheat_threshold: int = 3        # 过热阈值（遗物可升至 4）
var _is_overheated: bool = false
var _free_switches_this_turn: int = 1   # 每回合免费切换次数
var _switches_used_this_turn: int = 0

func on_battle_start() -> void:
    _current_mode = ArmorMode.ATTACK
    _switch_counter = 0
    _is_overheated = false
    _switches_used_this_turn = 0

func on_turn_start(_round_num: int) -> void:
    if _is_overheated:
        # 过热回合: 不可切换 + +2力+2敏
        _is_overheated = false
        if _resource_mgr:
            _resource_mgr.modify_player_strength(2)
            _resource_mgr.modify_player_dexterity(2)
    _switches_used_this_turn = 0

func on_turn_end() -> void:
    match _current_mode:
        ArmorMode.ATTACK:
            if _resource_mgr:
                _resource_mgr.modify_player_hp(-2, "normal")
        ArmorMode.DEFENSE:
            if _resource_mgr:
                _resource_mgr.modify_player_hp(2, "heal")

## 切换战甲模式
func switch_mode() -> void:
    if _is_overheated:
        return
    _switches_used_this_turn += 1
    if _switches_used_this_turn > _free_switches_this_turn:
        # 额外切换消耗精神力
        if _resource_mgr and _resource_mgr.get_player_energy() < 1:
            return
        if _resource_mgr:
            _resource_mgr.modify_player_energy(-1)

    _current_mode = ArmorMode.DEFENSE if _current_mode == ArmorMode.ATTACK else ArmorMode.ATTACK
    _switch_counter += 1

    if _switch_counter >= _overheat_threshold:
        _is_overheated = true
        _switch_counter = 0

func get_extra_damage(_target: Node) -> int:
    if _current_mode == ArmorMode.ATTACK:
        return 0  # +30% 由 get_damage_multiplier 处理
    return 0

func get_damage_multiplier(_target: Node) -> float:
    if _current_mode == ArmorMode.ATTACK:
        return 1.3
    return 1.0

func get_extra_block() -> int:
    return 0

func get_block_multiplier() -> float:
    if _current_mode == ArmorMode.DEFENSE:
        return 1.3
    return 1.0

func get_current_mode() -> ArmorMode:
    return _current_mode

## 遗物修正
func set_overheat_threshold(t: int) -> void:
    _overheat_threshold = t

func set_attack_mode_bonus(mult: float) -> void:
    _attack_bonus = mult

func set_defense_mode_bonus(mult: float) -> void:
    _defense_bonus = mult
```
